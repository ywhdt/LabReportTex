\makeatletter
\NewCommandCopy{\labreport@cref}{\cref}

\NewCommandCopy{\labreport@crefrange}{\crefrange}

\RenewDocumentCommand{\cref}{s m}{%
  \noeqref{#2}%
  \IfBooleanTF{#1}
    {\labreport@cref*{#2}}
    {\labreport@cref{#2}}}

\RenewDocumentCommand{\Cref}{s m}{%
  \noeqref{#2}%
  \IfBooleanTF{#1}
    {\labreport@Cref*{#2}}
    {\labreport@Cref{#2}}}

\RenewDocumentCommand{\crefrange}{s m m}{%
  \noeqref{#2,#3}%
  \IfBooleanTF{#1}
    {\labreport@crefrange*{#2}{#3}}
    {\labreport@crefrange{#2}{#3}}}

\RenewDocumentCommand{\Crefrange}{s m m}{%
  \noeqref{#2,#3}%
  \IfBooleanTF{#1}
    {\labreport@Crefrange*{#2}{#3}}
    {\labreport@Crefrange{#2}{#3}}}
\makeatother
